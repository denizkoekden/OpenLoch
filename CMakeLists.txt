cmake_minimum_required(VERSION 3.24)
project(OpenLoch VERSION 0.1.0 LANGUAGES CXX)
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets PrintSupport)
qt_standard_project_setup()
add_library(openloch_core src/legacy_reader.cpp src/legacy_writer.cpp src/project.cpp src/board.cpp src/geometry.cpp src/history.cpp src/notes.cpp src/continuity.cpp src/assistant.cpp src/hpgl.cpp src/language.cpp src/language_en.cpp src/language_fr.cpp src/installation.cpp src/partbuilder.cpp src/openlibrary.cpp src/icons.cpp src/printing.cpp src/picturefill.cpp src/documents/projectfile.cpp src/documents/targets.cpp src/targetcheck.cpp src/emfwriter.cpp)
target_include_directories(openloch_core PUBLIC src)
target_link_libraries(openloch_core PUBLIC Qt6::Core Qt6::Gui)
qt_add_executable(OpenLoch MACOSX_BUNDLE WIN32 src/main.cpp src/window.cpp src/canvas.cpp src/printdialog.cpp)
target_link_libraries(OpenLoch PRIVATE openloch_core Qt6::Widgets Qt6::PrintSupport)
qt_add_resources(OpenLoch "app_icon" PREFIX "/" BASE assets FILES assets/openloch.png)
# OpenLoch's own component libraries (CC0), generated into LIB pages at start.
file(GLOB OPENLOCH_LIBRARY_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/libraries/*.json")
qt_add_resources(OpenLoch "open_libraries" PREFIX "/" FILES ${OPENLOCH_LIBRARY_FILES})
# The help pages (Hilfe → Hilfethemen).
set(OPENLOCH_HELP_FILES help/de.html help/en.html help/fr.html)
qt_add_resources(OpenLoch "help" PREFIX "/" FILES ${OPENLOCH_HELP_FILES})
if(APPLE)
    set_source_files_properties(assets/OpenLoch.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(OpenLoch PRIVATE assets/OpenLoch.icns)
    set_target_properties(OpenLoch PROPERTIES MACOSX_BUNDLE_ICON_FILE OpenLoch.icns)
elseif(WIN32)
    enable_language(RC)
    configure_file(assets/OpenLoch.rc.in ${CMAKE_CURRENT_BINARY_DIR}/OpenLoch.rc @ONLY)
    target_sources(OpenLoch PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/OpenLoch.rc)
endif()
# Qt's template for the Info.plist with the document types the Finder opens with OpenLoch.
set_target_properties(OpenLoch PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME OpenLoch
    MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/assets/Info.plist.in
    MACOSX_BUNDLE_GUI_IDENTIFIER org.openloch.desktop
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION})
# README pictures from the program's own symbols and library (scripts/make-readme-art.py); not part of the normal build.
add_executable(openloch_readme_art EXCLUDE_FROM_ALL tools/readme_art.cpp src/canvas.cpp src/suite/documents.cpp src/suite/boardsources.cpp src/suite/schematictargets.cpp)
target_link_libraries(openloch_readme_art PRIVATE openloch_core openloch_schematic openloch_pcb openloch_frontpanel Qt6::Widgets Qt6::PrintSupport)
add_executable(openloch_inspect src/inspect.cpp)
target_link_libraries(openloch_inspect PRIVATE openloch_core)
enable_testing()
add_executable(openloch_tests tests/core_tests.cpp)
target_link_libraries(openloch_tests PRIVATE openloch_core)
target_compile_definitions(openloch_tests PRIVATE OPENLOCH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core COMMAND openloch_tests)
add_executable(openloch_writer_tests tests/writer_tests.cpp)
target_link_libraries(openloch_writer_tests PRIVATE openloch_core)
add_test(NAME writer COMMAND openloch_writer_tests)
add_executable(openloch_editor_tests tests/editor_tests.cpp src/canvas.cpp)
target_link_libraries(openloch_editor_tests PRIVATE openloch_core Qt6::Widgets Qt6::PrintSupport)
add_test(NAME editor COMMAND openloch_editor_tests)
set_tests_properties(editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(openloch_window_tests tests/window_tests.cpp src/window.cpp src/canvas.cpp src/printdialog.cpp)
target_link_libraries(openloch_window_tests PRIVATE openloch_core Qt6::Widgets Qt6::PrintSupport)
target_compile_definitions(openloch_window_tests PRIVATE OPENLOCH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
qt_add_resources(openloch_window_tests "help" PREFIX "/" FILES ${OPENLOCH_HELP_FILES})
add_test(NAME window COMMAND openloch_window_tests)
set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
# Each document module brings its own targets and tests in its module.cmake (see docs/suite.md).
include(${CMAKE_CURRENT_SOURCE_DIR}/src/modules/frontpanel/module.cmake OPTIONAL)
include(${CMAKE_CURRENT_SOURCE_DIR}/src/modules/pcb/module.cmake OPTIONAL)
include(${CMAKE_CURRENT_SOURCE_DIR}/src/modules/schematic/module.cmake OPTIONAL)
# The suite (src/suite/): start screen, kinds of document and the windows of the modules, in the program and its test.
set(OPENLOCH_SUITE_SOURCES src/suite/boardsources.cpp src/suite/documents.cpp src/suite/libraries.cpp src/suite/pcbwindow.cpp src/suite/projectoverview.cpp src/suite/projectparts.cpp src/suite/schematictargets.cpp src/suite/startscreen.cpp src/suite/suite.cpp)
target_sources(OpenLoch PRIVATE ${OPENLOCH_SUITE_SOURCES})
target_link_libraries(OpenLoch PRIVATE openloch_frontpanel openloch_pcb openloch_schematic)
add_executable(openloch_suite_tests tests/suite_tests.cpp ${OPENLOCH_SUITE_SOURCES} src/window.cpp src/canvas.cpp src/printdialog.cpp)
target_link_libraries(openloch_suite_tests PRIVATE openloch_frontpanel openloch_pcb openloch_schematic Qt6::Widgets Qt6::PrintSupport)
target_compile_definitions(openloch_suite_tests PRIVATE OPENLOCH_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME suite COMMAND openloch_suite_tests)
set_tests_properties(suite PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
# A test that waits for a dialog nobody answers fails after ten minutes instead of stalling the run (CTest's default is 25).
set_tests_properties(core writer editor window suite PROPERTIES TIMEOUT 600)
install(TARGETS OpenLoch BUNDLE DESTINATION . RUNTIME DESTINATION bin)
